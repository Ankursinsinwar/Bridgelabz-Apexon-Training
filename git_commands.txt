# git init: Initializes a local git repository.

# git clone: Creates a local copy of a remote repository.

# git fetch: Fetches the changes from the remote repository without merging them into the local repository.

# git remote: Lists all the remote repositories that are connected to the local repository.

# git remote add: connect local repositories to the remote repository.

# git add: Adds a file to the staging area.

# git restore --staged: remove the file from the staging area.

# git commit -m: Commits the changes in the staging area to the local repository.

# git commit --amend: Change the Most Recent Commit Message (Not Yet Pushed) [push with force (if previously pushed)]

# git rebase -i HEAD~n: Change an Older Commit (Not Yet Pushed)
		[Change pick to reword for the commit(s) you want to edit.
		Actions:
			pick   → keep commit
			reword → edit message
			squash → combine commits
			drop   → remove commit	]

# git revert abc1234(hash): Change the Commit Message (Pushed)
		[This creates a new commit that undoes the changes made by that commit
		 This does not change the original message but safely undoes the commit itself]

# git commit --allow-empty -m "new msg": This adds a no-op (empty) commit with a new message.


# git push: Pushes the changes in the local repository to the remote repository.

# git pull: Pulls the changes from the remote repository to the local repository.

# git branch: Lists all the branches in the local repository.

# git checkout: Switches to a different branch.

# git switch --orphan <new branch>: create empty branch

# git checkout --orphan <new branch>: create empty branch

# git checkout -b <new branch>: create branch

# git switch -c <new branch>: create branch

# git merge: Merges two branches together.

# git rebase: Rebases the changes in the local repository onto a different branch.

# git grep: Searches text in repository.

# git log: Shows the history of the commits in the local repository.

# git diff: Shows the difference between two commits.

# git status: Shows the status of the local repository.

# git stash: Stashes the changes in the working directory.

# git reset: Resets the working directory to a previous commit.

# git tag: Creates a tag in the local repository.

# git mv: Moves or renames a file and tracks it.

# git rm: Removes files from Git tracking.

# git rmdir: remove folder [must be empty].

# git submodule add: Adds another repo inside a repo.

GIT COMMANDS WITH EXAMPLE
------------------------------------------------------------------------

git init
Initializes a new local Git repository.
Example:
git init


git clone
Creates a local copy of a remote repository.
Example:
git clone https://github.com/user/repo.git


git remote
Lists connected remote repositories.
Example:
git remote
git remote -v


git remote add
Adds a remote repository.
Example:
git remote add origin https://github.com/user/repo.git

git fetch
Fetches changes from remote without merging.
Example:
git fetch origin


git status
Shows current state of the repository.
Example:
git status

git add
Adds files to staging area.
Example:
git add file.txt
git add .
git add src/

git commit
Commits staged changes.
Example:
git commit -m "Feat: Add login feature"
git commit -am "Fix(Login): forgot password fix"


git push
Pushes commits to remote.
Example:
git push origin main
git push -u origin main


git pull
Fetches and merges changes.
Example:
git pull origin main


git branch
Lists or manages branches.
Example:
git branch
git branch branch1
git branch -d branch1


git checkout
Switch branches or restore files.
Example:
git checkout branch1
git checkout -b branch1


git switch
Modern alternative to checkout.
Example:
git switch main
git switch -c branch1


git merge
Merges branches.
Example:
git merge branch1


git rebase
Reapplies commits on top of another branch.
Example:
git rebase main


git log
Shows commit history.
Example:
git log
git log --oneline
git log --graph --all


git diff
Shows differences between commits or files.
Example:
git diff
git diff HEAD~1


git show
Shows details of a commit.
Example:
git show commit_hash


git blame
Shows who modified each line.
Example:
git blame app.py


git stash
Temporarily saves changes.
Example:
git stash
git stash pop
git stash list


git reset
Moves HEAD to previous commit.
Example:
git reset --soft HEAD~1
git reset --hard HEAD~1


git revert
Creates a new commit that undoes a commit.
Example:
git revert commit_hash


git tag
Creates version tags.
Example:
git tag v1.0
git tag -a v1.1 -m "Stable release"
git push origin v1.0


git config
Sets Git configuration.
Example:
git config --global user.name "Your Name"
git config --global user.email "email@example.com"


git clean
Removes untracked files.
Example:
git clean -n
git clean -f


git cherry-pick
Applies a specific commit from another branch.
Example:
git cherry-pick commit_hash


git fetch --all
Fetches updates from all remotes.
Example:
git fetch --all


git pull --rebase
Pulls changes and rebases instead of merging.
Example:
git pull --rebase origin main


git push --force-with-lease
Safely force-pushes changes.
Example:
git push --force-with-lease origin main


git mv
Moves or renames a file and tracks it.
Example:
git mv old_name.py new_name.py


git rm
Removes files from Git tracking.
Example:
git rm file.txt
git rm --cached file.txt


git commit --amend
Edits the last commit.
Example:
git commit --amend -m "Updated commit message"


git log --author
Shows commits by a specific author.
Example:
git log --author="Ankur"


git log --since
Shows commits since a date.
Example:
git log --since="2 weeks ago"


git grep
Searches text in repository.
Example:
git grep "TODO"


git branch -a
Lists local and remote branches.
Example:
git branch -a


git branch -m
Renames a branch.
Example:
git branch -m old-name new-name


git prune
Removes stale remote-tracking branches.
Example:
git remote prune origin


git submodule add
Adds another repo inside a repo.
Example:
git submodule add https://github.com/user/lib.git


git rebase -i
Interactive rebase.
Example:
git rebase -i HEAD~5

Actions:
pick   → keep commit
reword → edit message
squash → combine commits
drop   → remove commit


git reflog
Shows all HEAD movements.
Example:
git reflog



git commit -S
Signs commits with GPG.
Example:
git commit -S -m "Secure commit"
